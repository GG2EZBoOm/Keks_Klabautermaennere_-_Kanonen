name post:ende
entry ps_main
target ps_5_0
flags 32768
macros 0
text 4446

Texture2D t0 : register(t0); Texture2D t1 : register(t1); Texture2D t2 : register(t2); Texture2D t3 : register(t3);
Texture2D t4 : register(t4); Texture2D t5 : register(t5); Texture2D t6 : register(t6); Texture2D t7 : register(t7);
SamplerState s_linear : register(s0);
SamplerState s_point : register(s1);

cbuffer Ende : register(b0) {
  float bloom, belichtung, saettigung, kontrast;
  float vignette, korn, zufall, weich;
  float strahlen, focusDistance, aperture, pad0;
  float2 bildgroesse, lensPixels;
  float2 nearFar, pad1;
  float3 gain; float pad2;
  float3 lift; float pad3;
  float3 shadowTint; float pad4;
  float3 highlightTint; float pad5;
};
// t0 tSzene, t1 tBloom, t2 tStrahlen, t3 tKontakt, t4 tTiefe (= lensDepth), t5 tAir
float viewDepth(float2 uv){
  float d=t4.SampleLevel(s_point,uv,0).r;
  return nearFar.x*nearFar.y/(nearFar.y-d*(nearFar.y-nearFar.x));
}
float3 lensColor(float2 uv){
  float centerDepth=viewDepth(uv);
  float radius=clamp(abs(1.0-focusDistance/max(centerDepth,.1))*aperture,0.0,5.0);
  float3 color=t0.SampleLevel(s_linear,uv,0).rgb;
  if(radius<.35)return color;
  float weights=1.0;
  [loop] for(int i=0;i<16;i++){
    float angle=float(i)*2.399963,ring=sqrt((float(i)+.5)/16.0);
    float2 sampleUV=uv+float2(cos(angle),sin(angle))*radius*ring/lensPixels;
    float sampleDepth=viewDepth(sampleUV);
    float sampleRadius=clamp(abs(1.0-focusDistance/max(sampleDepth,.1))*aperture,0.0,5.0);
    float w=smoothstep(radius*ring-.6,radius*ring+.6,sampleRadius);
    w*=sampleDepth<centerDepth?exp(-(centerDepth-sampleDepth)*.7):1.0;
    color+=t0.SampleLevel(s_linear,sampleUV,0).rgb*w;weights+=w;
  }
  return color/weights;
}
float4 atmosphereAt(float2 uv){
  float center=viewDepth(uv),weight=0.0;float4 color=float4(0,0,0,0);
  [unroll] for(int i=0;i<5;i++){
    float2 shift=i==0?float2(0,0):i==1?float2(1,0):i==2?float2(-1,0):i==3?float2(0,1):float2(0,-1);
    float2 q=uv+shift/lensPixels;
    float w=exp(-abs(viewDepth(q)-center)*.8);
    color+=t5.SampleLevel(s_linear,q,0)*w;weight+=w;
  }
  return color/max(weight,.001);
}
float3 aces(float3 x){
  const float a = 2.51, b = 0.03, c = 2.43, d = 0.59, e = 0.14;
  return clamp((x * (a * x + b)) / (x * (c * x + d) + e), 0.0, 1.0);
}
float rauschen(float2 p){ return frac(sin(dot(p, float2(12.9898, 78.233))) * 43758.5453); }
float3 zuSRGB(float3 c){
  return lerp(c * 12.92, 1.055 * pow(max(c, float3(0,0,0)), float3(1.0 / 2.4, 1.0 / 2.4, 1.0 / 2.4)) - 0.055, step(0.0031308, c));
}
float4 ps_main(float4 pos : SV_POSITION, float2 vUv : TEXCOORD0) : SV_Target {
  float3 c = lensColor(vUv);
  float depth=t4.SampleLevel(s_point,vUv,0).r, ao=0.0, weight=0.0;
  [unroll] for(int i=0;i<5;i++){
    float2 offset=i==0?float2(0,0):i==1?float2(2,0):i==2?float2(-2,0):i==3?float2(0,2):float2(0,-2);
    float2 uv=vUv+offset/bildgroesse;
    float w=exp(-abs(t4.SampleLevel(s_point,uv,0).r-depth)*1500.0);
    ao+=t3.SampleLevel(s_linear,uv,0).r*w;weight+=w;
  }
  c *= ao/max(weight,.001);
  float4 air=atmosphereAt(vUv);
  c=c*air.a+air.rgb;
  if (weich > 0.001) {
    float2 e = weich / bildgroesse;
    float3 u = t0.SampleLevel(s_linear, vUv + float2(e.x, 0.0), 0).rgb
             + t0.SampleLevel(s_linear, vUv - float2(e.x, 0.0), 0).rgb
             + t0.SampleLevel(s_linear, vUv + float2(0.0, e.y), 0).rgb
             + t0.SampleLevel(s_linear, vUv - float2(0.0, e.y), 0).rgb;
    c = lerp(c, (c * 1.5 + u) / 5.5, 0.55);
  }
  c += t1.SampleLevel(s_linear, vUv, 0).rgb * bloom;
  c += t2.SampleLevel(s_linear, vUv, 0).rgb * strahlen;
  c *= belichtung;
  float luminanz = max(dot(c, float3(0.2126, 0.7152, 0.0722)), 0.0001);
  c *= pow(luminanz / 0.18, kontrast - 1.0);
  c = c * gain;
  float tief = dot(c, float3(0.2126, 0.7152, 0.0722));
  c += lift * .18 * (1.0 - smoothstep(0.0, 0.22, tief));
  c *= lerp(shadowTint,float3(1,1,1),smoothstep(.015,.22,tief));
  c *= lerp(float3(1,1,1),highlightTint,smoothstep(.3,1.3,tief));
  float l = dot(c, float3(0.2126, 0.7152, 0.0722));
  c = lerp(float3(l,l,l), c, saettigung);
  c = aces(c);
  float2 d = (vUv - 0.5) * float2(1.0, 0.94);
  float v = 1.0 - vignette * pow(clamp(length(d) * 1.42, 0.0, 1.0), 2.4);
  c *= v;
  float mitte = dot(c, float3(0.2126, 0.7152, 0.0722));
  float kornAnteil = smoothstep(0.02, 0.35, mitte) * (1.0 - smoothstep(0.55, 1.0, mitte));
  c += (rauschen(vUv * 947.0 + zufall) - 0.5) * korn * kornAnteil;
  return float4(zuSRGB(max(c, float3(0,0,0))), 1.0);
}

Texture2D t0 : register(t0); Texture2D t1 : register(t1); Texture2D t2 : register(t2); Texture2D t3 : register(t3);
Texture2D t4 : register(t4); Texture2D t5 : register(t5); Texture2D t6 : register(t6); Texture2D t7 : register(t7);
SamplerState s_linear : register(s0);
SamplerState s_point : register(s1);

cbuffer Air : register(b0) {
    float4x4 inverseProjection, cameraWorld, shadowMatrix;
    float4 eye, lightDirection, lightColor, ambientColor;
    float density, groundLevel, interior, hasShadow;
};
// t0 depthMap, t1 sunShadow
float hash(float3 p){p=frac(p*.1031);p+=dot(p,p.yzx+33.33);return frac((p.x+p.y)*p.z);}
float noise3(float3 p){
  float3 i=floor(p),f=frac(p);f=f*f*(3.0-2.0*f);
  return lerp(lerp(lerp(hash(i),hash(i+float3(1,0,0)),f.x),lerp(hash(i+float3(0,1,0)),hash(i+float3(1,1,0)),f.x),f.y),
              lerp(lerp(hash(i+float3(0,0,1)),hash(i+float3(1,0,1)),f.x),lerp(hash(i+float3(0,1,1)),hash(i+float3(1,1,1)),f.x),f.y),f.z);
}
float visibility(float3 p){
  if(hasShadow<.5)return 1.0;
  float4 s=mul(shadowMatrix,float4(p,1.0));s.xyz/=s.w;
  if(any(s.xyz<0.0)||any(s.xyz>1.0))return 1.0;
  float result=0.0;
  [unroll] for(int k=0;k<4;k++){
    float2 o=float2(float(k%2)*2.0-1.0,float(k/2)*2.0-1.0)/3072.0;
    result+=step(s.z-.00012,t1.SampleLevel(s_point,s.xy+o,0).r);
  }
  return result*.25;
}
float windowBeam(float3 p,float x){
  float3 origin=float3(x,3.8,-10.6),axis=normalize(float3(1.5,-3.8,12.2));
  float3 delta=p-origin;float t=dot(delta,axis);
  if(t<0.0||t>22.0)return 0.0;
  float r=length(delta-axis*t),width=1.05+t*.045;
  return (1.0-smoothstep(width*.65,width,r))*(1.0-smoothstep(12.0,22.0,t));
}
float4 ps_main(float4 pos : SV_POSITION, float2 vUv : TEXCOORD0) : SV_Target {
  float depth=t0.SampleLevel(s_point,vUv,0).r;
  float4 p=mul(inverseProjection,float4(vUv*2.0-1.0,depth*2.0-1.0,1.0));p/=p.w;
  float3 endPoint=mul(cameraWorld,p).xyz,ray=normalize(endPoint-eye.xyz);
  float lengthRay=min(length(endPoint-eye.xyz),interior>.5?36.0:115.0);
  float stepLength=lengthRay/24.0;
  float jitter=hash(float3(pos.xy,7.13));
  float transmission=1.0;float3 scatter=float3(0,0,0);
  float facing=max(dot(ray,lightDirection.xyz),0.0);
  float phase=.32+1.8*pow(facing,4.0);
  [loop] for(int i=0;i<24;i++){
    float3 world=eye.xyz+ray*(float(i)+jitter)*stepLength;
    float heightFade=exp(-max(world.y-groundLevel,0.0)*.11);
    float variation=.55+.85*noise3(world*float3(.065,.13,.065));
    float localDensity=density*heightFade*variation;
    float3 illumination=ambientColor.rgb;
    if(interior>.5){
      float beam=windowBeam(world,-12.0)+windowBeam(world,-9.0);
      localDensity+=beam*.001;
      illumination+=float3(.21,.30,.40)*beam;
    }else illumination+=lightColor.rgb*visibility(world)*phase;
    float absorbed=1.0-exp(-localDensity*stepLength);
    scatter+=transmission*absorbed*illumination;
    transmission*=1.0-absorbed;
  }
  return float4(scatter,transmission);
}

Texture2D t0 : register(t0); Texture2D t1 : register(t1); Texture2D t2 : register(t2); Texture2D t3 : register(t3);
Texture2D t4 : register(t4); Texture2D t5 : register(t5); Texture2D t6 : register(t6); Texture2D t7 : register(t7);
SamplerState s_linear : register(s0);
SamplerState s_point : register(s1);

cbuffer Kontakt : register(b0) { float4x4 inverseProjection; float2 pixel; };
// t0 tTiefe
float3 positionAt(float2 uv) {
  float d=t0.SampleLevel(s_point,uv,0).x;
  float4 p=mul(inverseProjection,float4(uv*2.0-1.0,d*2.0-1.0,1.0));
  return p.xyz/p.w;
}
float4 ps_main(float4 pos : SV_POSITION, float2 vUv : TEXCOORD0) : SV_Target {
  float depth=t0.SampleLevel(s_point,vUv,0).x;
  if(depth>.99998)return float4(1,1,1,1);
  float3 p=positionAt(vUv);
  float3 left=p-positionAt(vUv-float2(pixel.x*4.0,0.0));
  float3 right=positionAt(vUv+float2(pixel.x*4.0,0.0))-p;
  float3 down=p-positionAt(vUv-float2(0.0,pixel.y*4.0));
  float3 up=positionAt(vUv+float2(0.0,pixel.y*4.0))-p;
  float3 normal=normalize(cross(abs(left.z)<abs(right.z)?left:right,abs(down.z)<abs(up.z)?down:up));
  float occlusion=0.0;
  float radius=clamp(36.0/max(-p.z,1.0),.6,5.0);
  float rotation=frac(dot(floor(vUv/pixel),float2(.754877,.56984)))*6.283185;
  [loop] for(int i=0;i<12;i++){
    float angle=float(i)*2.39996+rotation;
    float2 uv=vUv+float2(cos(angle),sin(angle))*pixel*(3.0+float(i)*1.55)*radius;
    float3 delta=positionAt(clamp(uv,float2(0.001,0.001),float2(.999,.999)))-p;
    float distance=length(delta);
    occlusion+=max(dot(normal,delta)/max(distance,.001)-.10,0.0)*(1.0-smoothstep(.08,1.5,distance));
  }
  return float4(clamp(1.0-occlusion*.19,.40,1.0).xxx,1.0);
}
